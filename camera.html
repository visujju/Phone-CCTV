// Ensure canvas doesn't crash if filter is applied function renderVideoWithTimestamp() { if (rawVideo.readyState >= rawVideo.HAVE_CURRENT_DATA) { try { if (isNightVision) { canvasCtx.filter = 'brightness(180%) contrast(160%) grayscale(80%)'; } else { canvasCtx.filter = 'none'; } canvasCtx.drawImage(rawVideo, 0, 0, streamCanvas.width, streamCanvas.height); canvasCtx.filter = 'none'; // Always reset filter const now = new Date(); const dateStr = now.toLocaleDateString(); const timeStr = now.toLocaleTimeString(); const timeStampText = `${dateStr} ${timeStr}`; canvasCtx.fillStyle = 'rgba(0, 0, 0, 0.6)'; canvasCtx.fillRect(20, 20, 380, 50); canvasCtx.fillStyle = '#34c759'; canvasCtx.font = 'bold 24px monospace'; canvasCtx.fillText(timeStampText, 35, 53); } catch (err) { console.error('Canvas render error:', err); } } requestAnimationFrame(renderVideoWithTimestamp); }
